================================================================================
1. GENEL BAKIŞ - DBE ANALİZİ NEDİR?
================================================================================

DBE (Degree of Unsaturation / Index of Hydrogen Deficiency):
- Moleküldeki doymamışlık derecesini hesaplayan bir formül
- Aromatik halkalar, çift bağlar, üçlü bağlar, halkalar için kullanılır
- Formül: DBE = C - H/2 - X/2 + N/2 + 1
- Nerede: 
  * C = Karbon sayısı
  * H = Hidrojen sayısı
  * N = Azot sayısı
  * X = Halojenler (F + Cl + Br + I)

Kod tarafında üretilen format:
```
**0. DBE Analizi (ZORUNLU!):**
- Formül: C47H51NO14
- DBE Hesabı: DBE = 47 - 51/2 - 0/2 + 1/2 + 1 = 23
- DBE Yorumu: DBE≥4: aromatik/çoklu halka veya çoklu doymamışlık olasılığı artar (kompleks yapı olabilir).
```

```typescript
  const normalizedFormula = normalizeFormula(effectiveFormula);
  const dbeBlock = renderDBEBlock(normalizedFormula);
  
  if (dbeBlock) {
    // Replace existing DBE block if present
    // Pattern: "**0. DBE Analizi" ... until next "**" header (like "**1.") or end

    if (dbeBlockStartPattern.test(result.reasoning)) {
      // Find the start of DBE block
      const dbeStartMatch = result.reasoning.match(dbeBlockStartPattern);
      if (dbeStartMatch) {
        const dbeStartIndex = dbeStartMatch.index!;

        const nextHeaderMatch = afterDbeStart.match(/\n\n\*\*[^0]/);

          // Replace from DBE start to next header
          const dbeEndIndex = dbeStartIndex + nextHeaderMatch.index!;
          result.reasoning = 
            result.reasoning.substring(0, dbeStartIndex) +
            dbeBlock +
            result.reasoning.substring(dbeEndIndex);
        } else {
          // Replace from DBE start to end
          result.reasoning = 
            result.reasoning.substring(0, dbeStartIndex) +
            dbeBlock;
        }
      }
      console.log(`✅ DBE block replaced with locked formula: ${normalizedFormula}`);
    } else {
      // Insert DBE block at the beginning (before "**1." or at start)
      const firstSectionIndex = result.reasoning.indexOf('**1.');
      if (firstSectionIndex > 0) {
        result.reasoning = dbeBlock + '\n\n' + result.reasoning.substring(firstSectionIndex);
      } else {
        result.reasoning = dbeBlock + '\n\n' + result.reasoning;
      }
      console.log(`✅ DBE block inserted with locked formula: ${normalizedFormula}`);
```

```typescript
export function renderDBEBlock(formula: string): string {
  const { hillFormula, atomCounts } = parseFormula(formula);
  const { dbe, terms } = computeDBEDetailed(atomCounts);

  // Format DBE nicely: integer if close, else keep one decimal
  const rounded = Math.round(dbe * 2) / 2; // to nearest 0.5
  const dbeStr = Number.isInteger(rounded) ? String(rounded) : String(rounded);

  // Build a deterministic "calculation string"
  // DBE = C - H/2 - X/2 + N/2 + 1
  let calc = `DBE = ${terms.C}`;
  if (terms.H > 0) calc += ` - ${terms.H}/2`;
  if (terms.X > 0) calc += ` - ${terms.X}/2`;
  if (terms.N > 0) calc += ` + ${terms.N}/2`;
  calc += ` + 1 = ${dbeStr}`;

  // Minimal interpretation (avoid hallucinations)
  let interp = "";
  if (rounded <= 0) interp = "DBE≈0: doymuş yapı (çift bağ/halka beklenmez).";
  else if (rounded === 1) interp = "DBE≈1: 1 halka veya 1 çift bağ (veya 1 karbonil) olası.";
  else if (rounded >= 4) interp = `DBE≥4: aromatik/çoklu halka veya çoklu doymamışlık olasılığı artar (kompleks yapı olabilir).`;
  else interp = `DBE=${dbeStr}: birden fazla doymamışlık/halka olası.`;

  return [
    `**0. DBE Analizi (ZORUNLU!):**`,
    `- Formül: ${hillFormula || formula}`,
    `- DBE Hesabı: ${calc}`,
    `- DBE Yorumu: ${interp}`,
  ].join("\n");
}
```

Değişkenler:

- dbe: number (hesaplanan DBE değeri)
- terms: {C, H, N, X} (DBE hesaplama terimleri)
- rounded: number (0.5'e yuvarlanmış DBE)

```typescript
export function parseFormula(formula: string): ParseFormulaResult {
  const normalizedInput = stripCommonNoise(formula);

  if (!normalizedInput) {
    return { normalizedInput, atomCounts: {}, hillFormula: "" };
  }

  // Tokenize as repeating groups: Element + optional count
  // Example: C47 H51 N O14
  const re = /([A-Z][a-z]?)(\d*)/g;

  const atomCounts: AtomCounts = {};
  let match: RegExpExecArray | null;
  let consumed = 0;

  while ((match = re.exec(normalizedInput)) !== null) {
    const [full, element, countStr] = match;
    consumed += full.length;

    if (!isElementSymbol(element)) continue;

    const count = countStr ? Number.parseInt(countStr, 10) : 1;
    if (!Number.isFinite(count) || count <= 0) continue;

    atomCounts[element] = (atomCounts[element] ?? 0) + count;
  }

  const hillFormula = formatHillFormula(atomCounts);

  return { normalizedInput, atomCounts, hillFormula };
}
```

Algoritma:
1. stripCommonNoise(): Unicode subscript'leri, boşlukları, charge'ları temizler
2. Regex ile element + sayı tokenize eder: /([A-Z][a-z]?)(\d*)/g
3. Her element için atom sayısını toplar
4. formatHillFormula() ile Hill sisteminde formatlar

Değişkenler:
- normalizedInput: string (temizlenmiş formül)
- atomCounts: AtomCounts (örn: {C:47, H:51, N:1, O:14})
- hillFormula: string (örn: "C47H51NO14")

```typescript
export function computeDBEDetailed(atomCounts: AtomCounts, charge: number = 0): DBEResult {
  const C = atomCounts['C'] ?? 0;
  const H = atomCounts['H'] ?? 0;
  const N = atomCounts['N'] ?? 0;

  const X =
    (atomCounts['F'] ?? 0) +
    (atomCounts['Cl'] ?? 0) +
    (atomCounts['Br'] ?? 0) +
    (atomCounts['I'] ?? 0);

  const dbe = C - H / 2 - X / 2 + N / 2 + charge / 2 + 1;

  return { dbe, terms: { C, H, N, X } };
}
```

Formül:
DBE = C - H/2 - X/2 + N/2 + charge/2 + 1

Nerede:
- C = Karbon sayısı
- H = Hidrojen sayısı
- N = Azot sayısı
- X = Halojenler (F + Cl + Br + I)
- charge = Molekül yükü (varsayılan: 0)

Örnek Hesaplama: C47H51NO14 (Paclitaxel)
- C = 47
- H = 51
- N = 1
- X = 0 (halojen yok)
- charge = 0
- DBE = 47 - 51/2 - 0/2 + 1/2 + 0/2 + 1 = 47 - 25.5 + 0.5 + 1 = 23

Değişkenler:

- charge: number (varsayılan: 0)
- dbe: number (hesaplanan DBE değeri)
- terms: {C, H, N, X} (DBE hesaplama terimleri)

```typescript
export function renderDBEBlock(formula: string): string {
  const { hillFormula, atomCounts } = parseFormula(formula);
  const { dbe, terms } = computeDBEDetailed(atomCounts);

  const rounded = Math.round(dbe * 2) / 2;
  const dbeStr = Number.isInteger(rounded) ? String(rounded) : String(rounded);

  let calc = `DBE = ${terms.C}`;
  if (terms.H > 0) calc += ` - ${terms.H}/2`;
  if (terms.X > 0) calc += ` - ${terms.X}/2`;
  if (terms.N > 0) calc += ` + ${terms.N}/2`;
  calc += ` + 1 = ${dbeStr}`;

  let interp = "";
  if (rounded <= 0) interp = "DBE≈0: doymuş yapı (çift bağ/halka beklenmez).";
  else if (rounded === 1) interp = "DBE≈1: 1 halka veya 1 çift bağ (veya 1 karbonil) olası.";
  else if (rounded >= 4) interp = `DBE≥4: aromatik/çoklu halka veya çoklu doymamışlık olasılığı artar (kompleks yapı olabilir).`;
  else interp = `DBE=${dbeStr}: birden fazla doymamışlık/halka olası.`;

  return [
    `**0. DBE Analizi (ZORUNLU!):**`,
    `- Formül: ${hillFormula || formula}`,
    `- DBE Hesabı: ${calc}`,
    `- DBE Yorumu: ${interp}`,
  ].join("\n");
}
```

- terms: {C, H, N, X} (DBE hesaplama terimleri)
- rounded: number (0.5'e yuvarlanmış DBE)
- dbeStr: string (formatlanmış DBE string'i)
- calc: string (hesaplama string'i)
- interp: string (minimal yorum)
